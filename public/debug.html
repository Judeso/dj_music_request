🔗 Problème: Event Listener Manquant

Si le bouton n'a pas d'événement attaché :

// Ajouter ceci dans votre script.js document.addEventListener('DOMContentLoaded', function() { const createButton = document.getElementById('create-event-btn'); if (createButton) { createButton.addEventListener('click', function(e) { e.preventDefault(); console.log('Bouton cliqué!'); createEvent(); }); } else { console.error('Bouton non trouvé!'); } });

function validateEventData(eventData) { if (!eventData.name || eventData.name.trim() === '') { alert('Le nom de l\'événement est requis'); return false; } if (!eventData.date) { alert('La date est requise'); return false; } return true; }
🚀 Code Complet Fonctionnel
async function createEvent() { try { // Récupérer les données du formulaire const eventData = { name: document.getElementById('event-name').value, type: document.getElementById('event-type').value, date: document.getElementById('event-date').value, location: document.getElementById('event-location').value, expectedGuests: parseInt(document.getElementById('event-guests').value), status: 'preparation' }; // Valider les données if (!eventData.name || !eventData.date || !eventData.location) { alert('Veuillez remplir tous les champs requis'); return; } // Afficher loading const button = document.getElementById('create-event-btn'); button.disabled = true; button.textContent = 'Création...'; // Envoyer à l'API const response = await fetch('/.netlify/functions/events', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(eventData) }); const result = await response.json(); if (result.success) { alert('Événement créé avec succès!'); location.reload(); // Recharger la page } else { alert('Erreur: ' + result.error); } } catch (error) { console.error('Erreur:', error); alert('Erreur de connexion'); } finally { // Remettre le bouton en état const button = document.getElementById('create-event-btn'); button.disabled = false; button.textContent = 'Créer Événement'; } }
